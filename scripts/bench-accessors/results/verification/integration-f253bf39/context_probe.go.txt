package main

import (
	"context"
	"encoding/json"
	"os"
	"time"
)

func main() {
	for i := 0; i < 10; i++ {
		parent, cancelParent := context.WithTimeout(context.Background(), 50*time.Millisecond)
		// Siblings widen the documented interval between parent Done closing and
		// propagation to a child; the Go fuzzer has the same parent/child layout.
		for n := 0; n < 10000; n++ {
			context.WithCancel(parent)
		}
		child, cancelChild := context.WithCancel(parent)
		<-parent.Done()
		err, before := parent.Err(), child.Err()
		// The actual suppression condition in Go 1.25.1 internal/fuzz/fuzz.go.
		if err == before {
			err = nil
		}
		cancelChild()
		json.NewEncoder(os.Stdout).Encode(map[string]any{
			"iteration": i + 1, "parent_error": label(parent.Err()),
			"child_error_before_stop": label(before),
			"fuzzer_condition_retains_error": label(err),
			"child_error_after_stop": label(child.Err()),
		})
		cancelParent()
	}
}

func label(err error) any {
	if err == nil {
		return nil
	}
	return err.Error()
}
