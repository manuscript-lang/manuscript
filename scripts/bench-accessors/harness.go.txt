var accessorResult rt.Value

type accessorCase struct {
 name string
 fn rt.Function
 receiver rt.Value
 perIteration int64
}

 table:=&rt.ListTable[int64]{}

// Preserve checked arithmetic, loop polling, native receiver admission,
// source-function admission/delivery and the authoritative mutable List.
func handwrittenLength() rt.Function {
 return rt.NewSourceFunction([]string{"values","limit"},"Int","accessor-benchmark.ms:1:1",func(ctx *rt.Execution,args []rt.Value) rt.Value {
  values,limit:=args[0],rt.IntValue(args[1])
  var total,index int64
  var tick uint32
  for index<limit {
   rt.Poll(ctx,&tick)
   total=rt.IntAdd(total,rt.CollectionLengthInt(ctx,values))
   index=rt.IntAdd(index,1)
  }
  return total
 })
}

func TestAccessorParity(t *testing.T) {
 initialize()
 ctx:=rt.RootExecution()
 defer rt.FinishExecution(ctx)
 for _,limit:=range []int64{0,1,32,1024} {
  for _,test:=range accessorCases() {
   want:=test.perIteration*limit
   if test.name=="StringScalars/Alternating" {want=(limit/2)*7+(limit%2)*5}
   if got:=rt.IntValue(test.fn.Body(ctx,[]rt.Value{test.receiver,limit}));got!=want {
    t.Fatalf("%s(%d)=%d, want %d",test.name,limit,got,want)
   }
  }
 }
 for _,test:=range accessorCases() {
  if test.name!="ListLength" {continue}
  for _,limit:=range []int64{0,1,32,1024} {
   args:=[]rt.Value{test.receiver,limit}
   if got,want:=handwrittenLength().Body(ctx,args),test.fn.Body(ctx,args);got!=want {t.Fatalf("typed helper: %v != %v",got,want)}
  }
 }
}

func TestAccessorCancellation(t *testing.T) {
 initialize()
 ctx:=rt.RootExecution()
 defer rt.FinishExecution(ctx)
 var cancel context.CancelFunc
 ctx.Context,cancel=context.WithCancel(ctx.Context)
 cancel()
 for _,test:=range accessorCases() {
  func(){
   defer func(){if recover()==nil {t.Errorf("%s ignored source admission cancellation",test.name)}}()
   test.fn.Body(ctx,[]rt.Value{test.receiver,int64(1024)})
  }()
 }
}

func BenchmarkAccessor(b *testing.B) {
 initialize()
 ctx:=rt.RootExecution()
 defer rt.FinishExecution(ctx)
 for _,test:=range accessorCases() {
  b.Run(test.name,func(b *testing.B){
   args:=[]rt.Value{test.receiver,int64(1024)}
   b.ReportAllocs()
   for b.Loop(){accessorResult=test.fn.Body(ctx,args)}
  })
  if test.name=="ListLength" {
   b.Run("HandwrittenTypedHelper",func(b *testing.B){
    fn:=handwrittenLength()
    args:=[]rt.Value{test.receiver,int64(1024)}
    b.ReportAllocs()
    for b.Loop(){accessorResult=fn.Body(ctx,args)}
   })
  }
 }
}
