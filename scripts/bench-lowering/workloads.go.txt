var benchmarkResult rt.Value
var nativeResult int64

func goScalar(limit int64) int64 {var total int64;for i:=int64(0);i<limit;i++ {total+=i%7};return total}
//go:noinline
func goAdd(a,b int64) int64 {return a+b}
func goCalls(limit int64) int64 {var total int64;for i:=int64(0);i<limit;i++ {total=goAdd(total,i%7)};return total}
func goItems(values []int64,limit int64)int64 {var total int64;for i:=int64(0);i<limit;i++ {total+=values[i%int64(len(values))]};return total}
func goLife(side,generations int64)int64 {
 cells:=make([]int64,side*side)
 for i:=range cells {if i%7==0 {cells[i]=1}}
 for generation:=int64(0);generation<generations;generation++ {
  next:=make([]int64,len(cells))
  for i:=int64(0);i<side*side;i++ {
   row,col:=i/side,i%side;north,south:=(row+side-1)%side,(row+1)%side;west,east:=(col+side-1)%side,(col+1)%side
   neighbours:=cells[north*side+west]+cells[north*side+col]+cells[north*side+east]+cells[row*side+west]+cells[row*side+east]+cells[south*side+west]+cells[south*side+col]+cells[south*side+east]
   if neighbours==3 || neighbours==2 && cells[i]==1 {next[i]=1}
  }
  cells=next
 }
 var alive int64;for _,v:=range cells {alive+=v};return alive
}
func TestBenchmarkParity(t *testing.T) {
 initialize();ctx:=rt.RootExecution();defer rt.FinishExecution(ctx)
 for _,n:=range []int64{0,1,32,1024} {
  want:=goScalar(n)
  for _,f:=range []rt.Value{@@scalar@@,@@calls@@} {if got:=f.(rt.Function).Body(ctx,[]rt.Value{n});got!=want {t.Fatalf("loop %v != %v",got,want)}}
 }
 values:=&rt.List{Type:"@std/List[Int]",Items:[]rt.Value{int64(10000),int64(20000),int64(30000),int64(40000)}}
 for _,f:=range []rt.Value{@@items@@,@@matched@@} {if got:=f.(rt.Function).Body(ctx,[]rt.Value{values,int64(1024)});got!=int64(25600000) {t.Fatalf("access %v",got)}}
 if got:=@@ranged@@.(rt.Function).Body(ctx,[]rt.Value{values});got!=int64(100000) {t.Fatal("range",got)}
 for _,f:=range []rt.Value{@@sort_values@@,@@sort_captured@@} {if got:=f.(rt.Function).Body(ctx,[]rt.Value{values});got!=int64(10000) {t.Fatal("sort",got)}}
 for _,side:=range []int64{3,8,16,128} {for _,generations:=range []int64{0,1,3,50} {
  want:=goLife(side,generations);got:=@@life@@.(rt.Function).Body(ctx,[]rt.Value{side,generations})
  if got!=want {t.Fatalf("life(%d,%d) %v != %d",side,generations,got,want)}
 }}
 boxed:=[]rt.Value{int64(10000)}
 for _,test:=range []struct{name string;fn rt.Value;want int64}{{"Identity",@@identity@@,10000},{"Forwarded",@@forwarded@@,10000},{"Copied",@@copied@@,10000},{"Capture",@@capture@@,10001},{"Stored",@@stored@@,10001}} {
  if got:=test.fn.(rt.Function).Body(ctx,boxed);got!=test.want {t.Fatalf("%s: %v",test.name,got)}
 }
}
