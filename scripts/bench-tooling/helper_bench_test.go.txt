// This file is supplied through go -overlay; it never changes the checkout.
var reviewOccurrences []occurrence
var reviewSemantic map[string]any
var reviewPosition position
var reviewOffset int

func reviewSource(n int) string {
	var text strings.Builder
	for i := 0; i < n; i++ {
		fmt.Fprintf(&text, "fn f%06d(n: Int) Int: n + 1\n", i)
	}
	return text.String()
}

func BenchmarkReviewIndex(b *testing.B) {
	for _, n := range []int{0, 1000, 2000, 4000} {
		b.Run(fmt.Sprint(n), func(b *testing.B) {
			source := reviewSource(n)
			result := compiler.CompileSource("scale.ms", source, compiler.Options{AnalyzeOnly: true})
			if !result.Success() {
				b.Fatalf("fixture failed: %v", result.Diagnostics)
			}
			index := sourceIndex(result)
			b.Logf("source_bytes=%d modules=%d occurrence_count=%d", len(source), len(sourceModules(result)), len(index))
			b.ReportAllocs()
			b.ResetTimer()
			for i := 0; i < b.N; i++ {
				reviewOccurrences = sourceIndex(result)
			}
		})
	}
}

	for _, n := range []int{1000, 2000, 4000} {

func BenchmarkReviewSemantic(b *testing.B) {
	for _, n := range []int{1000, 2000, 4000, 8000} {
		b.Run(fmt.Sprint(n), func(b *testing.B) {
			source := reviewSource(n)
			d := document{URI: "file:///scale.ms", Text: source}
			// Omit semantic checking to isolate lexing plus byte-to-UTF16 mapping.
			b.SetBytes(int64(len(source)))
			b.ReportAllocs()
			b.ResetTimer()
			for i := 0; i < b.N; i++ {
				reviewSemantic = semantic(d)
			}
		})
	}
}

func BenchmarkReviewCheckedSemantic(b *testing.B) {
	path := os.Getenv("MANUSCRIPT_REVIEW_SEMANTIC_FILE")
	if path == "" {
		b.Skip("set MANUSCRIPT_REVIEW_SEMANTIC_FILE to an existing checked project module")
	}
	source, err := os.ReadFile(path)
	if err != nil {
		b.Fatal(err)
	}
	d := document{URI: fileURI(path), Text: string(source), Version: 1}
	result := analyze(d, map[string]document{d.URI: d})
	if !result.Success() {
		b.Fatal(result.Diagnostics)
	}
	b.Logf("source_bytes=%d project_modules=%d occurrence_count=%d", len(source), len(result.Program.Modules), len(sourceIndex(result)))
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		reviewSemantic = semantic(d, result)
	}
}

func BenchmarkReviewLongLine(b *testing.B) {
	for _, n := range []int{128, 1024, 8192} {
		for _, unit := range []string{"a", "😀"} {
			b.Run(fmt.Sprintf("%s/%d", unit, n), func(b *testing.B) {
				source := strings.Repeat(unit, n)
				b.ReportAllocs()
				b.ResetTimer()
				for i := 0; i < b.N; i++ {
					reviewPosition = Position(source, len(source))
				}
			})
		}
	}
}

func TestReviewPositionContracts(t *testing.T) {
	for _, source := range []string{"alpha\nbeta\n", "α😀\r\nβ😀\n", ""} {
		for at := 0; at <= len(source); at++ {
			if at < len(source) && source[at]&0xc0 == 0x80 {
				continue
			}
			if at > 0 && source[at-1] == '\r' {
				continue
			}
			p := Position(source, at)
			got, err := Offset(source, p)
			if err != nil || got != at {
				t.Fatalf("roundtrip source=%q at=%d p=%+v got=%d err=%v", source, at, p, got, err)
			}
		}
	}
	if _, err := Offset("😀", position{Character: 1}); err == nil {
		t.Fatal("surrogate-split negative control accepted")
	}
}
