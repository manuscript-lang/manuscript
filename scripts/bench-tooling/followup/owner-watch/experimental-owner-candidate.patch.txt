+// Ownership records selection before compilation, including failed results.
+// They contain paths, never Programs or a second dependency graph. Immutable
+// standard documents deliberately have no local project ownership.
+type analysisOwner struct {
+	target, root, canonical string
+	known, immutable        bool
+}
+
+func localFilePath(uri string) (string, bool) {
+	u, err := url.Parse(uri)
+	if err != nil || u.Scheme != "file" || u.Opaque != "" || u.RawQuery != "" || u.Fragment != "" || u.Host != "" && u.Host != "localhost" {
+		return "", false
+	}
+	path := filename(uri)
+	return path, filepath.IsAbs(path)
+}
+
+func targetOwner(target string) analysisOwner {
+	root := filepath.Dir(target)
+	canonical, err := filepath.EvalSymlinks(root)
+	return analysisOwner{target: target, root: root, canonical: canonical, known: err == nil}
+}
+
+func documentOwner(d document) analysisOwner {
+	if isStandardLibraryURI(d.URI) || standardLibraryDocument(d) != nil {
+		return analysisOwner{known: true, immutable: true}
+	}
+	path, ok := localFilePath(d.URI)
+	if !ok {
+		return analysisOwner{}
+	}
+	target, found := project.FindManifest(path)
+	if !found {
+		target = path
+	}
+	return targetOwner(target)
+}

+// Capture the target analyzeWith actually selects, rather than selecting a
+// manifest independently before analysis. The callback also records ownership
+// for failed compilations. Immutable documents bypass compilation entirely.
+func (s *server) documentCompiler(d document) func(string, compiler.Options) compiler.Result {
+	if s.owners == nil {
+		s.owners = map[string]analysisOwner{}
+	}
+	if isStandardLibraryURI(d.URI) || standardLibraryDocument(d) != nil {
+		s.owners[d.URI] = analysisOwner{known: true, immutable: true}
+	}
+	_, local := localFilePath(d.URI)
+	share := s.canShareOverlays()
+	return func(target string, options compiler.Options) compiler.Result {
+		owner := analysisOwner{}
+		if local {
+			owner = targetOwner(target)
+		}
+		s.owners[d.URI] = owner
+		if share {
+			return s.compileSnapshot(target, options, owner)
+		}
+		return compiler.CompileFile(target, options)
+	}
+}

+type watchedPath struct {
+	written, canonical  string
+	manifest, directory bool

+// A deleted path still has a useful physical footprint if an existing ancestor
+// resolves. A dangling existing symlink or other filesystem uncertainty falls
+// back to full invalidation; this never admits an overlay or an import.
+func eventCanonical(path string) (string, bool) {
+	current := path
+	var suffix []string
+	for {
+		canonical, err := filepath.EvalSymlinks(current)
+		if err == nil {
+			for i := len(suffix) - 1; i >= 0; i-- {
+				canonical = filepath.Join(canonical, suffix[i])
+			}
+			return canonical, true
+		}
+		if !os.IsNotExist(err) {
+			return "", false
+		}
+		if _, err := os.Lstat(current); err == nil || !os.IsNotExist(err) {
+			return "", false
+		}
+		parent := filepath.Dir(current)
+		if parent == current {
+			return "", false
+		}
+		suffix = append(suffix, filepath.Base(current))
+		current = parent
+	}
+}

+func watchedPaths(params json.RawMessage) ([]watchedPath, bool) {
+	var envelope struct {
+		Changes []struct {
+			URI string `json:"uri"`
+		} `json:"changes"`
+	}
+	if json.Unmarshal(params, &envelope) != nil || envelope.Changes == nil {
+		return nil, false
+	}
+	var paths []watchedPath
+	for _, event := range envelope.Changes {
+		if isStandardLibraryURI(event.URI) {
+			continue
+		}
+		path, ok := localFilePath(event.URI)
+		if !ok {
+			return nil, false
+		}
+		manifest := filepath.Base(path) == "ms.toml"
+		info, err := os.Stat(path)
+		if err != nil && !os.IsNotExist(err) {
+			return nil, false
+		}
+		// Missing paths no longer reveal their former kind or lexical symlink
+		// targets. A received event beneath an owner may represent deletion of
+		// an imported directory, regardless of its suffix; refresh that owner.
+		directory := err != nil || info.IsDir()
+		if !manifest && !directory && filepath.Ext(path) != ".ms" {
+			continue
+		}
+		canonical, ok := eventCanonical(path)
+		if !ok {
+			return nil, false
+		}
+		paths = append(paths, watchedPath{path, canonical, manifest, directory})
+	}
+	return paths, true
+}

+func (o analysisOwner) matches(event watchedPath) bool {
+	if o.immutable {
+		return false
+	}
+	for _, root := range []string{o.root, o.canonical} {
+		for _, path := range []string{event.written, event.canonical} {
+			ancestor := path
+			if event.manifest {
+				ancestor = filepath.Dir(path)
+			}
+			if watchPathWithin(root, path) || (event.directory || event.manifest) && watchPathWithin(ancestor, root) {
+				return true
+			}
+		}
+	}
+	return false
+}

+// invalidateWatched returns every affected open URI, even if its analysis was
+// already evicted. Unknown metadata retains the previous safe global fallback.
+func (s *server) invalidateWatched(paths []watchedPath) (map[string]bool, bool) {
+	current := map[string]analysisOwner{}
+	affectedRoots := map[string]bool{}
+	mark := func(owner analysisOwner) {
+		if !owner.immutable {
+			affectedRoots[owner.canonical] = true
+		}
+	}
+	for uri, d := range s.docs {
+		owner := documentOwner(d)
+		if !owner.known {
+			return nil, false
+		}
+		current[uri] = owner
+		old, exists := s.owners[uri]
+		if _, cached := s.cache[uri]; cached && (!exists || !old.known) {
+			return nil, false
+		}
+		if exists && old != owner {
+			mark(old)
+			mark(owner)
+		}
+		for _, event := range paths {
+			if owner.matches(event) || exists && old.matches(event) {
+				mark(owner)
+				if exists {
+					mark(old)
+				}
+			}
+		}
+	}
+	for key := range s.analyses {
+		owner, exists := s.analysisOwners[key]
+		if !exists || !owner.known {
+			return nil, false
+		}
+		for _, event := range paths {
+			if owner.matches(event) {
+				mark(owner)
+			}
+		}
+	}
+	affected := map[string]bool{}
+	for uri, owner := range current {
+		old := s.owners[uri]
+		if !owner.immutable && (affectedRoots[owner.canonical] || old.known && affectedRoots[old.canonical]) {
+			affected[uri] = true
+			delete(s.cache, uri)
+			delete(s.owners, uri)
+		}
+	}
+	for key, owner := range s.analysisOwners {
+		if affectedRoots[owner.canonical] {
+			delete(s.analyses, key)
+			delete(s.analysisOwners, key)
+		}
+	}
+	// A retained entry can still refer to a dropped entry's Program. Only release
+	// derived state after checking all remaining live owners.
+	live := map[*checker.Program]bool{}
+	for _, result := range s.cache {
+		live[result.Program] = true
+	}
+	for _, result := range s.analyses {
+		live[result.Program] = true
+	}
+	for program := range s.indices {
+		if !live[program] {
+			delete(s.indices, program)
+		}
+	}
+	for program := range s.lints {
+		if !live[program] {
+			delete(s.lints, program)
+		}
+	}
+	s.overlayChecked, s.shareOverlays = false, false
+	return affected, true
+}
+
+func (s *server) watchedFiles(params json.RawMessage) error {
+	// A pending didChange publication is an obligation for every open URI.
+	if err := s.flushDiagnostics(); err != nil {
+		return err
+	}
+	paths, understood := watchedPaths(params)
+	if understood && len(paths) == 0 {
+		return nil
+	}
+	if understood {
+		if affected, ok := s.invalidateWatched(paths); ok {
+			return s.publishSelected(affected)
+		}
+	}
+	s.invalidate()
+	return s.publish()
+}
+
+func watchPathWithin(root, path string) bool {
+	rel, err := filepath.Rel(root, path)
+	return err == nil && !filepath.IsAbs(rel) && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator))
+}

+func TestWatchRecordsActuallySelectedManifestAcrossTopologyChange(t *testing.T) {
+	root, files := snapshotFiles(t, map[string]string{
+		"ms.toml":        "[project]\nname = \"parent\"\nsource = \".\"\nentry = \"main.ms\"\n",
+		"main.ms":        "import nested from \"./nested/d.ms\"\nfn main() Int: nested()\n",
+		"dep.ms":         "export fn value() Int: 1\n",
+		"nested/ms.toml": "[project]\nname = \"nested\"\nsource = \".\"\nentry = \"d.ms\"\n",
+		"nested/d.ms":    "import value from \"../dep.ms\"\nexport fn nested() Int: value()\n",
+	})
+	d := files["nested/d.ms"]
+	s := snapshotServer(d)
+	manifest := filepath.Join(root, "nested/ms.toml")
+	// Construct the ordinary production compile callback while B exists, then
+	// change topology before analyzeWith selects its manifest. This deterministic
+	// interleaving distinguishes actual selection from independent preselection.
+	compile := s.documentCompiler(d)
+	if err := os.Remove(manifest); err != nil {
+		t.Fatal(err)
+	}
+	before := analyzeWith(d, s.docs, compiler.Options{AnalyzeOnly: true, Recover: true}, compile)
+	if !before.Success() || before.ProjectRoot != root || s.owners[d.URI].target != filepath.Join(root, "ms.toml") {
+		t.Fatal("URI ownership did not capture actual parent selection", before.Diagnostics, s.owners[d.URI])
+	}
+	s.cache[d.URI] = before
+	watchWrite(t, manifest, "[project]\nname = \"nested\"\nsource = \".\"\nentry = \"d.ms\"\n")
+	watchWrite(t, filepath.Join(root, "dep.ms"), "export fn value() String: \"changed\"\n")
+	// Only the parent dependency event has arrived so far; current B selection
+	// differs from actual cached A even though the earlier preselection was B.
+	watchSend(t, s, fileURI(filepath.Join(root, "dep.ms")))
+	if got := snapshotEquivalent(t, s, d); got.Success() || got.Program == before.Program || s.owners[d.URI].target != manifest {
+		t.Fatal("parent analysis survived restored nested selection", got.Diagnostics, s.owners[d.URI])
+	}
