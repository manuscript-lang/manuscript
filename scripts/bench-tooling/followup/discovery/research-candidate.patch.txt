+	"context"

+type discoveryScope struct {
+	root      string
+	recursive bool
+}
+
+// Only completed physical walks are shared, and only within one request.
+// Documents retain their written paths so analysis still selects their nearest
+// manifest before canonicalizing its containing directory.
+type workspaceDiscovery struct {
+	walks    map[discoveryScope][]string
+	contents map[string]string
+}
+
+func (w *workspaceDiscovery) disk(root string, recursive bool, canceled func() bool) ([]string, error) {
+	key := discoveryScope{root, recursive}
+	if files, ok := w.walks[key]; ok {
+		return files, nil
+	}
+	var files []string
+	err := filepath.WalkDir(root, func(path string, entry fs.DirEntry, err error) error {
+		if isCanceled(canceled) {
+			return context.Canceled
+		}
+		if err != nil {
+			return err
+		}
+		if entry.IsDir() {
+			if path != root && (!recursive || strings.HasPrefix(entry.Name(), ".") || entry.Name() == "node_modules" || entry.Name() == "vendor") {
+				return filepath.SkipDir
+			}
+			return nil
+		}
+		if entry.Type()&os.ModeSymlink != 0 || filepath.Ext(path) != ".ms" {
+			return nil
+		}
+		files = append(files, path)
+		return nil
+	})
+	if err != nil {
+		if os.IsNotExist(err) {
+			// Preserve the old partial-result behavior if files disappear, but
+			// do not reuse an incomplete traversal in another written context.
+			return files, nil
+		}
+		return nil, err
+	}
+	if w.walks == nil {
+		w.walks = map[discoveryScope][]string{}
+	}
+	w.walks[key] = files
+	return files, nil
+}
+
+func pathWithin(root, path string) bool {
+	rel, err := filepath.Rel(root, path)
+	return err == nil && !filepath.IsAbs(rel) && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator))
+}
+
+func (w *workspaceDiscovery) documents(d document, docs map[string]document, recursive bool, canceled func() bool) ([]document, error) {
+	if isCanceled(canceled) {
+		return nil, context.Canceled
+	}
+	if isStandardLibraryURI(d.URI) || standardLibraryDocument(d) != nil {
+		return []document{d}, nil
+	}
+	writtenRoot, recursive := workspaceScope(d, recursive)
+	root := writtenRoot
+	if canonical, err := filepath.EvalSymlinks(root); err == nil {
+		root = canonical
+	}
+	disk, err := w.disk(root, recursive, canceled)
+	if err != nil {
+		return nil, err
+	}
+	// SourcePath needs all written overlay keys to recognize confined new files.
+	// Text is selected below from actual URI records, never a map-order winner.
+	overlay := map[string]string{filename(d.URI): d.Text}
+	for _, open := range docs {
+		if isCanceled(canceled) {
+			return nil, context.Canceled
+		}
+		if !isStandardLibraryURI(open.URI) && standardLibraryDocument(open) == nil {
+			overlay[filename(open.URI)] = open.Text
+		}
+	}
+	files := map[string]document{}
+	represented := map[string]bool{}
+	texts := map[string]string{}
+	ambiguous := map[string]bool{}
+	add := func(open document) {
+		if isStandardLibraryURI(open.URI) || standardLibraryDocument(open) != nil {
+			return
+		}
+		path := filename(open.URI)
+		canonical, err := project.SourcePath(root, path, overlay)
+		if !pathWithin(writtenRoot, path) && err != nil {
+			return
+		}
+		files[open.URI] = open
+		represented[path] = true
+		if err == nil {
+			if text, exists := texts[canonical]; exists && text != open.Text {
+				ambiguous[canonical] = true
+			}
+			texts[canonical] = open.Text
+		}
+	}
+	for _, open := range docs {
+		if isCanceled(canceled) {
+			return nil, context.Canceled
+		}
+		if open.URI != d.URI {
+			add(open)
+		}
+	}
+	add(d)
+	files[d.URI] = d
+	for _, physical := range disk {
+		if isCanceled(canceled) {
+			return nil, context.Canceled
+		}
+		relative, _ := filepath.Rel(root, physical)
+		written := filepath.Join(writtenRoot, relative)
+		if represented[written] || ambiguous[physical] {
+			continue
+		}
+		text, overlaid := texts[physical]
+		if !overlaid {
+			var cached bool
+			text, cached = w.contents[physical]
+			if !cached {
+				data, err := os.ReadFile(physical)
+				if err != nil {
+					if os.IsNotExist(err) {
+						continue
+					}
+					return nil, err
+				}
+				text = string(data)
+				if w.contents == nil {
+					w.contents = map[string]string{}
+				}
+				w.contents[physical] = text
+			}
+		}
+		uri := fileURI(written)
+		files[uri] = document{URI: uri, Text: text}
+	}
+	result := make([]document, 0, len(files))
+	for _, file := range files {
+		result = append(result, file)
+	}
+	sort.Slice(result, func(i, j int) bool { return result[i].URI < result[j].URI })
+	return result, nil
+}

+func TestDiscoverySharesPhysicalWalkButRetainsWrittenPaths(t *testing.T) {
+	root, _ := snapshotFiles(t, map[string]string{
+		"ms.toml": "[project]\nname = \"discovery\"\nsource = \".\"\nentry = \"main.ms\"\n",
+		"main.ms": "fn main() Int: 1\n", "unopened.ms": "fn unopened() Int: 2\n",
+	})
+	alias := discoveryAlias(t, root)
+	for _, roots := range [][]string{{alias, root}, {root, alias}} {
+		walk := &workspaceDiscovery{}
+		for _, written := range roots {
+			d := document{URI: fileURI(filepath.Join(written, "__workspace_index.ms"))}
+			files, err := walk.documents(d, nil, true, nil)
+			if err != nil {
+				t.Fatal(err)
+			}
+			found := map[string]bool{}
+			for _, file := range files {
+				path := filename(file.URI)
+				if !pathWithin(written, path) {
+					t.Fatal("walk changed lexical selection context", written, file.URI)
+				}
+				found[filepath.Base(path)] = true
+			}
+			if !found["main.ms"] || !found["unopened.ms"] {
+				t.Fatal("alias walk missed unopened source", found)
+			}
+		}
+		if len(walk.walks) != 1 {
+			t.Fatal("equivalent roots retained duplicate physical walks", len(walk.walks))
+		}
+		if symbols := workspaceSymbols(nil, roots, ""); len(symbols) != 2 {
+			t.Fatal("root order changed complete/deduplicated symbols", roots, symbols)
+		}
+	}
