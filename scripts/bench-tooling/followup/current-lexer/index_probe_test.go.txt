// lexicalInventorySnapshot is supplied by a separate baseline/candidate adapter.

func lexicalFixture(t testing.TB) (compiler.Result, string, string) {
	t.Helper()
	n, _ := strconv.Atoi(os.Getenv("LEXICAL_MODULES"))
	if n == 0 {
		n = 41
	}
	functions := 20
	root := t.TempDir()
	src := filepath.Join(root, "src")
	if err := os.Mkdir(src, 0700); err != nil {
		t.Fatal(err)
	}
	write := func(name, text string) {
		t.Helper()
		if err := os.WriteFile(filepath.Join(root, name), []byte(text), 0600); err != nil {
			t.Fatal(err)
		}
	}
	manifest := "[project]\nname = \"lexical_probe\"\nsource = \"src\"\n"
	if os.Getenv("LEXICAL_LIBRARY") != "1" {
		manifest += "entry = \"src/main.ms\"\n"
	}
	write("ms.toml", manifest)
	var main strings.Builder
	for module := 0; module < n-1; module++ {
		fmt.Fprintf(&main, "import f_%d_0 from \"./part_%d.ms\"\n", module, module)
		var source strings.Builder
		fmt.Fprintf(&source, "# module %d Unicode 😀 é\n", module)
		for fn := 0; fn < functions; fn++ {
			fmt.Fprintf(&source, "export fn f_%d_%d(value: Int) Int:\n    return value + %d\n", module, fn, fn)
		}
		write(fmt.Sprintf("src/part_%d.ms", module), source.String())
	}
	main.WriteString("\nexport fn main() Int:\n    let result = f_0_0(1)\n    return result\n")
	write("src/main.ms", main.String())
	result := compiler.CompileFile(filepath.Join(root, "ms.toml"), compiler.Options{AnalyzeOnly: true, Recover: true})
	if !result.Success() || result.Program == nil || len(result.Program.Modules) != n {
		t.Fatal("invalid lexical fixture", result.Diagnostics)
	}
	// These warm checked pinned source inventories equally without calling any LSP lexical inventory.
	standard := checker.StandardModules()
	if len(standard) == 0 {
		t.Fatal("missing pinned standard modules")
	}
	sourceInventory := make(map[string]string, len(standard))
	for _, m := range standard {
		sourceInventory[m.File] = m.Source
	}
	return result, root, lexicalHash(sourceInventory)
}
func lexicalHash(v any) string {
	data, err := json.Marshal(v)
	if err != nil {
		panic(err)
	}
	sum := sha256.Sum256(data)
	return fmt.Sprintf("%x", sum)
}
func lexicalNormalized(root, text string) string { return strings.ReplaceAll(text, root, "$FIXTURE") }

// Syntax facts carry checker cycles; serialize the full source shape with only
// those annotation fields removed. No pointer value is ever formatted.
func lexicalSyntax(v reflect.Value) any {
	if !v.IsValid() {
		return nil
	}
	if v.Kind() == reflect.Interface || v.Kind() == reflect.Pointer {
		if v.IsNil() {
			return nil
		}
		return lexicalSyntax(v.Elem())
	}
	switch v.Kind() {
	case reflect.Struct:
		m := map[string]any{}
		typ := v.Type()
		for i := 0; i < v.NumField(); i++ {
			f := typ.Field(i)
			if f.PkgPath == "" && f.Name != "Fact" {
				m[f.Name] = lexicalSyntax(v.Field(i))
			}
		}
		return m
	case reflect.Slice, reflect.Array:
		if v.Kind() == reflect.Slice && v.IsNil() {
			return nil
		}
		a := make([]any, v.Len())
		for i := range a {
			a[i] = lexicalSyntax(v.Index(i))
		}
		return a
	case reflect.Map:
		if v.IsNil() {
			return nil
		}
		m := map[string]any{}
		iter := v.MapRange()
		for iter.Next() {
			m[fmt.Sprint(iter.Key().Interface())] = lexicalSyntax(iter.Value())
		}
		return m
	default:
		return v.Interface()
	}
}

type lexicalSemantic struct {
	root  string
	types map[*checker.Type]int
	nodes []any
	decls map[*syntax.Declaration]any
}

func (c *lexicalSemantic) declaration(d *syntax.Declaration) any {
	if d == nil {
		return nil
	}
	if value, ok := c.decls[d]; ok {
		return value
	}
	raw := lexicalSyntax(reflect.ValueOf(d))
	data, _ := json.Marshal(raw)
	normalized := lexicalNormalized(c.root, string(data))
	sum := sha256.Sum256([]byte(normalized))
	value := map[string]any{"kind": d.Kind, "name": d.Name, "position": d.Pos, "export": d.Export, "constant": d.Constant, "synthetic": d.Synthetic, "source_shape_sha256": fmt.Sprintf("%x", sum)}
	c.decls[d] = value
	return value
}
func (c *lexicalSemantic) typ(t *checker.Type) any {
	if t == nil {
		return nil
	}
	if id, ok := c.types[t]; ok {
		return map[string]int{"ref": id}
	}
	id := len(c.nodes)
	c.types[t] = id
	c.nodes = append(c.nodes, nil)
	node := map[string]any{"id": t.ID, "name": t.Name, "kind": t.Kind, "args": c.typeList(t.Args), "result": c.typ(t.Result), "bounds": c.typeList(t.Bounds), "default": c.typ(t.Default), "defaults": t.Defaults}
	params := make([]any, len(t.Params))
	for i, p := range t.Params {
		params[i] = map[string]any{"name": p.Name, "type": c.typ(p.Type)}
	}
	node["params"] = params
	if d := t.Def; d != nil {
		fields := make([]any, len(d.Fields))
		for i, p := range d.Fields {
			fields[i] = map[string]any{"name": p.Name, "type": c.typ(p.Type)}
		}
		cases := make([]any, len(d.Cases))
		for i, cs := range d.Cases {
			fs := make([]any, len(cs.Fields))
			for j, f := range cs.Fields {
				fs[j] = map[string]any{"name": f.Name, "type": c.typ(f.Type)}
			}
			cases[i] = map[string]any{"name": cs.Name, "fields": fs}
		}
		operations := make([]any, len(d.Operations))
		for i, op := range d.Operations {
			operations[i] = map[string]any{"name": op.Name, "type": c.typ(op.Type), "declaration": c.declaration(op.Decl)}
		}
		node["definition"] = map[string]any{"kind": d.Kind, "symbol": c.symbol(d.Symbol), "variables": c.typeList(d.Variables), "fields": fields, "cases": cases, "operations": operations}
	}
	c.nodes[id] = node
	return map[string]int{"ref": id}
}
func (c *lexicalSemantic) typeList(types []*checker.Type) []any {
	out := make([]any, len(types))
	for i, t := range types {
		out[i] = c.typ(t)
	}
	return out
}
func (c *lexicalSemantic) symbol(s *checker.Symbol) any {
	if s == nil {
		return nil
	}
	return map[string]any{"name": s.Name, "id": s.ID, "kind": s.Kind, "module": s.Module, "native": s.Native, "export": s.Export, "type": c.typ(s.Type), "variables": c.typeList(s.Variables), "declaration": c.declaration(s.Decl), "is_constant": s.IsConstant(), "value_type": c.typ(s.ValueType())}
}
func (c *lexicalSemantic) binding(b *checker.Binding) any {
	if b == nil {
		return nil
	}
	return map[string]any{"name": b.Name, "id": b.ID, "type": c.typ(b.Type), "mutable": b.Mutable, "function": b.Function}
}
func (c *lexicalSemantic) method(m *checker.Method) any {
	if m == nil {
		return nil
	}
	return map[string]any{"id": m.ID, "module": m.Module, "local": m.Local, "receiver": c.typ(m.Receiver), "contract": c.typ(m.Contract), "type": c.typ(m.Type), "variables": c.typeList(m.Variables), "declaration": c.declaration(m.Decl), "native": m.Native}
}
func (c *lexicalSemantic) info(i *checker.Info) any {
	if i == nil {
		return nil
	}
	captures := make([]any, len(i.Captures))
	for j, b := range i.Captures {
		captures[j] = c.binding(b)
	}
	return map[string]any{"Type": c.typ(i.Type), "Symbol": c.symbol(i.Symbol), "Binding": c.binding(i.Binding), "Constructor": c.typ(i.Constructor), "Case": i.Case, "Method": c.method(i.Method), "Native": i.Native, "NoReturn": i.NoReturn, "Breaks": i.Breaks, "Write": i.Write, "Captures": captures, "TypeArgs": c.typeList(i.TypeArgs), "ErrorSource": c.typ(i.ErrorSource), "ScopedError": i.ScopedError, "PropagatedError": c.typ(i.PropagatedError), "ScopedPropagation": i.ScopedPropagation, "Arguments": lexicalSyntax(reflect.ValueOf(i.Arguments)), "OptionalArgument": c.typ(i.OptionalArgument), "Folded": i.Folded, "Constant": i.Constant, "Proven": i.Proven}
}
func lexicalPayload(t testing.TB, index []occurrence, root string) []byte {
	t.Helper()
	c := lexicalSemantic{root: root, types: map[*checker.Type]int{}, decls: map[*syntax.Declaration]any{}}
	records := make([]any, len(index))
	for n, e := range index {
		records[n] = map[string]any{"key": e.key, "name": e.name, "position": e.pos, "end": e.end, "declaration": e.declaration, "write": e.write, "info": c.info(e.info)}
	}
	data, err := json.Marshal(map[string]any{"occurrences": records, "types": c.nodes})
	if err != nil {
		t.Fatal(err)
	}
	return []byte(lexicalNormalized(root, string(data)))
}
func TestLexicalSerializerCoversInfo(t *testing.T) {
	c := lexicalSemantic{types: map[*checker.Type]int{}, decls: map[*syntax.Declaration]any{}}
	fields := c.info(&checker.Info{}).(map[string]any)
	typ := reflect.TypeOf(checker.Info{})
	for n := 0; n < typ.NumField(); n++ {
		field := typ.Field(n)
		if field.PkgPath == "" {
			if _, ok := fields[field.Name]; !ok {
				t.Fatalf("unserialized exported Info field %s", field.Name)
			}
		}
	}
}

var lexicalIndexSink []occurrence

func BenchmarkLexicalSourceIndex(b *testing.B) {
	result, root, sourceDigest := lexicalFixture(b)
	lexicalIndexSink = sourceIndex(result)
	payload := lexicalPayload(b, lexicalIndexSink, root)
	digest := sha256.Sum256(payload)
	b.Logf("std_source_sha256=%s occurrences=%d full_payload_sha256=%x", sourceDigest, len(lexicalIndexSink), digest)
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		lexicalIndexSink = sourceIndex(result)
	}
	b.StopTimer()
	runtime.KeepAlive(result)
}
func TestLexicalColdIndex(t *testing.T) {
	result, _, sourceDigest := lexicalFixture(t)
	runtime.GC()
	var before, after, retained runtime.MemStats
	runtime.ReadMemStats(&before)
	started := time.Now()
	index := sourceIndex(result)
	elapsed := time.Since(started)
	runtime.ReadMemStats(&after)
	count := len(index)
	if count == 0 {
		t.Fatal("empty source index")
	}
	index = nil
	runtime.GC()
	runtime.ReadMemStats(&retained)
	// Inventory inspection and semantic serialization stay outside all cold cost measurements.
	inventory := lexicalInventorySnapshot()
	record := map[string]any{"elapsed_ns": elapsed.Nanoseconds(), "allocated_bytes": after.TotalAlloc - before.TotalAlloc, "allocations": after.Mallocs - before.Mallocs, "retained_heap_delta": int64(retained.HeapAlloc) - int64(before.HeapAlloc), "retained_objects_delta": int64(retained.HeapObjects) - int64(before.HeapObjects), "occurrences": count, "std_source_sha256": sourceDigest, "inventory": inventory, "token_slot_size": unsafe.Sizeof(lexer.Token{}), "go_version": runtime.Version()}
	data, _ := json.Marshal(record)
	t.Logf("lexical_cold=%s", data)
	runtime.KeepAlive(result)
}

// This summary only reads the pinned inventory.
func lexicalPinnedSourceSummary() (int, int) {
	modules := checker.StandardModules()
	bytes := 0
	for _, m := range modules {
		bytes += len(m.Source)
	}
	return len(modules), bytes
}

func TestLexicalSourceIndexPayload(t *testing.T) {
	result, root, sourceDigest := lexicalFixture(t)
	index := sourceIndex(result)
	payload := lexicalPayload(t, index, root)
	sum := sha256.Sum256(payload)
	if dir := os.Getenv("LEXICAL_PAYLOAD_DIR"); dir != "" {
		if err := os.MkdirAll(dir, 0700); err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(filepath.Join(dir, "source-index.json"), payload, 0600); err != nil {
			t.Fatal(err)
		}
	}
	data, _ := json.Marshal(map[string]any{"occurrences": len(index), "std_source_sha256": sourceDigest, "payload_sha256": fmt.Sprintf("%x", sum), "payload_bytes": len(payload)})
	t.Logf("lexical_payload=%s", data)
}
func TestLexicalPinnedTokenInventory(t *testing.T) {
	inventory := map[string][]lexer.Token{}
	metrics := lexicalInventoryMetrics{}
	for _, m := range checker.StandardModules() {
		tokens, err := lexer.Lex(m.File, m.Source)
		if err != nil {
			t.Fatal(err)
		}
		inventory[m.File] = tokens
		metrics.Files++
		metrics.Tokens += len(tokens)
		metrics.TokenCapacity += cap(tokens)
		metrics.SourceBytes += len(m.Source)
	}
	metrics.TokenSlotBytes = metrics.TokenCapacity * int(unsafe.Sizeof(lexer.Token{}))
	data, _ := json.Marshal(map[string]any{"inventory": metrics, "token_payload_sha256": lexicalHash(inventory)})
	t.Logf("lexical_tokens=%s", data)
}
