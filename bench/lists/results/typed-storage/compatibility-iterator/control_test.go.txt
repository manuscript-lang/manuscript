package msruntime_test

import (
 "testing"
 rt "github.com/manuscript-lang/manuscript/pkg/runtime"
)

var compatibilityIteratorSink rt.Value

func compatibilityIteratorSum(ctx *rt.Execution, list *rt.List) rt.Value {
 var total int64
 for _, value := range rt.Iterate(list) {
  rt.CheckCancelled(ctx)
  total = rt.IntAdd(total,value.(int64))
 }
 return total
}

func BenchmarkCompatibilityIterator(b *testing.B) {
 ctx:=rt.RootExecution()
 defer rt.FinishExecution(ctx)
 items:=make([]rt.Value,65536)
 for i:=range items { items[i]=int64(i)*7919%100003 }
 list:=&rt.List{Type:"@std/List[Int]",Items:items}
 b.Run("DirectBLoop",func(b *testing.B){
  b.ReportAllocs()
  for b.Loop() {
   var total int64
   for _, value:=range rt.Iterate(list) {rt.CheckCancelled(ctx);total=rt.IntAdd(total,value.(int64))}
   compatibilityIteratorSink=total
  }
 })
 b.Run("FactoredProduction",func(b *testing.B){
  b.ReportAllocs()
  for b.Loop() {compatibilityIteratorSink=compatibilityIteratorSum(ctx,list)}
 })
}
