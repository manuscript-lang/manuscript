#!/usr/bin/env python3
"""Retain CLI wall/CPU/fault samples and exact outputs for effect probes."""

import argparse
import gzip
import hashlib
import json
import os
from pathlib import Path
import platform
import statistics
import subprocess
import time

from generate import source


def digest(data):
    return hashlib.sha256(data).hexdigest()


def run(command, prefix):
    with prefix.with_suffix(".stdout").open("wb") as out, prefix.with_suffix(".stderr").open("wb") as err:
        start = time.perf_counter_ns()
        child = subprocess.Popen(command, stdout=out, stderr=err)
        _, status, usage = os.wait4(child.pid, 0)
        elapsed = time.perf_counter_ns() - start
        child.returncode = os.waitstatus_to_exitcode(status)
    stdout = prefix.with_suffix(".stdout").read_bytes()
    stderr = prefix.with_suffix(".stderr").read_bytes()
    result = dict(command=command, wall_ns=elapsed, user_s=usage.ru_utime,
                  system_s=usage.ru_stime, minor_faults=usage.ru_minflt,
                  major_faults=usage.ru_majflt, max_rss_kib=usage.ru_maxrss,
                  exit=child.returncode, stdout_sha256=digest(stdout),
                  stderr_sha256=digest(stderr))
    if child.returncode != 0:
        raise RuntimeError(f"failed: {command}: {stderr.decode(errors='replace')}")
    return result, stdout, stderr


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("output", type=Path)
    parser.add_argument("--binary", action="append", required=True, help="label=/absolute/path")
    parser.add_argument("--sizes", default="1,20,250,500,1000")
    parser.add_argument("--modes", default="none,allowed,pure,direct")
    parser.add_argument("--commands", default="check,lint")
    parser.add_argument("--repeats", type=int, default=7)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    cpu = min(os.sched_getaffinity(0))
    os.sched_setaffinity(0, {cpu})
    binaries = [item.split("=", 1) for item in args.binary]
    metadata = dict(platform=platform.platform(), cpu=cpu, argv=os.sys.argv,
                    revision=subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
                    binaries={label: dict(path=path, sha256=digest(Path(path).read_bytes()))
                              for label, path in binaries})
    (args.output / "metadata.json").write_text(json.dumps(metadata, indent=2) + "\n")
    rows = []
    with (args.output / "samples.jsonl").open("w") as samples:
        for size in map(int, args.sizes.split(",")):
            for mode in args.modes.split(","):
                case = f"{mode}-{size}"
                input_file = (args.output / f"{case}.ms").resolve()
                input_file.write_text(source(size, mode))
                for command in args.commands.split(","):
                    expected = None
                    for trial in range(-1, args.repeats):
                        ordered = binaries if trial % 2 == 0 else list(reversed(binaries))
                        for label, binary in ordered:
                            prefix = args.output / f"{case}-{command}-{label}-{trial}"
                            row, stdout, stderr = run([binary, command, str(input_file), "--json"], prefix)
                            if expected is None:
                                expected = (stdout, stderr)
                                for suffix, data in (("stdout", stdout), ("stderr", stderr)):
                                    (args.output / f"{case}-{command}.{suffix}.gz").write_bytes(gzip.compress(data, mtime=0))
                            if expected != (stdout, stderr):
                                raise RuntimeError(f"output mismatch: {prefix}")
                            document = json.loads(stdout)
                            diagnostics = document["diagnostics"]
                            assert not any(d.get("severity") == "error" for d in diagnostics), prefix
                            count = sum(d.get("code") == "MSL116" for d in diagnostics)
                            wanted = {"none": 2 * size + 1, "direct": 2 * size + 1,
                                      "sparse": 1, "many": size, "cyclic": 2 * size}.get(mode, 0) if command == "lint" else 0
                            assert count == wanted, (prefix, count, wanted)
                            row.update(case=case, size=size, mode=mode, action=command,
                                       label=label, trial=trial, output_prefix=f"{case}-{command}",
                                       input_sha256=digest(input_file.read_bytes()))
                            samples.write(json.dumps(row) + "\n")
                            samples.flush()
                            rows.append(row)
                            prefix.with_suffix(".stdout").unlink()
                            prefix.with_suffix(".stderr").unlink()
                    for label, _ in binaries:
                        times = [r["wall_ns"] / 1e6 for r in rows if r["case"] == case
                                 and r["action"] == command and r["label"] == label and r["trial"] >= 0]
                        print(case, command, label, f"median {statistics.median(times):.3f} ms", flush=True)


if __name__ == "__main__":
    main()
