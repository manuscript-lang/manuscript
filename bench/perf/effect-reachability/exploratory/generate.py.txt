#!/usr/bin/env python3
"""Generate an unmeasured effect-path scaling probe and neighboring controls."""

import argparse
from pathlib import Path


def source(size: int, control: str) -> str:
    lines = []
    if control == "cyclic":
        for i in range(size):
            lines += ["// ms: effect probe.io", f"fn root_{i:06d}() Int:", f"    return leaf_{i:06d}()", ""]
            lines += [f"fn leaf_{i:06d}() Int:", f"    return root_{i:06d}()", ""]
        lines += ["fn main() Int:", "    return 0", ""]
        return "\n".join(lines)
    if control == "many":
        for i in range(size):
            lines += [f"// ms: effect probe.e{i:06d}", f"fn sink_{i:06d}() Int:", "    return 1", ""]
            lines += [f"fn root_{i:06d}() Int:", f"    return sink_{i:06d}()", ""]
        lines += ["fn main() Int:", "    return root_000000() - 1", ""]
        return "\n".join(lines)
    if control in ("allowed", "sparse"):
        lines += ["// ms: allow_effect probe.io", ""]
    if control != "pure":
        lines += ["// ms: effect probe.io"]
    lines += ["fn sink() Int:", "    return 1", ""]
    for i in range(size):
        lines += [f"fn leaf_{i:06d}() Int:", "    return sink()", ""]
    lines += ["fn hub() Int:"]
    lines += [f"    leaf_{i:06d}()" for i in range(size)]
    lines += ["    return 1", ""]
    for i in range(size):
        if control == "sparse" and i == 0:
            lines += ["// ms: allow_effect probe.none"]
        target = "sink" if control == "direct" or (control == "sparse" and i == 0) else "hub"
        lines += [f"fn root_{i:06d}() Int:", f"    return {target}()", ""]
    lines += ["fn main() Int:", "    return root_000000() - 1", ""]
    return "\n".join(lines)


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("size", type=int)
    parser.add_argument("output", type=Path)
    parser.add_argument("--control", choices=["none", "allowed", "pure", "direct", "sparse", "many", "cyclic"], default="none")
    args = parser.parse_args()
    if args.size < 1:
        parser.error("size must be positive")
    args.output.write_text(source(args.size, args.control))
