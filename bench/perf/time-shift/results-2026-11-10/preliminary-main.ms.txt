import (
	Timestamp, nanoseconds, monotonic, parse_timestamp from "@std/time.ms",
	arguments from "@std/os/env.ms",
)

type Input:
	time: Timestamp
	delta: Int

fn inputs(kind: String, count: Int) List[Input]:
	var result: List[Input] = []

	for i in 0..count:
		var t = Timestamp(seconds: 1700000000 + i, nanos: i * 7919 % 1000000000)
		var delta = (i % 17 - 8) * 123456789
		if kind == "zero": delta = 0
		if kind == "normalize": t.nanos = -1
		if kind == "wide": delta = if i % 2 == 0: 9223372036854775807 else: -9223372036854775807 - 1

		if kind == "fallback":
			t = Timestamp(seconds: 0, nanos: 9223372036854775807)
			delta = 9223372036854775807

		if kind == "error":
			t = Timestamp(seconds: 253402300799, nanos: 999999999)
			delta = 1

		result.append(Input(time: t, delta: delta))

	return result

fn work(kind: String, data: List[Input], rounds: Int) Int!:
	var checksum = 0

	for repeat in 0..rounds:
		for item in data:
			if kind == "parse":
				t := try parse_timestamp("2026-03-01T12:30:45.123Z")
				checksum += t.seconds + t.nanos
			else if kind == "format":
				checksum += item.time.to_string().length()
			else if kind == "error":
				result := item.time.shift(nanoseconds(item.delta))
				if result.is_ok() or result.err().kind != "ERANGE":
					return error(Error("expected ERANGE"))

				checksum += 1
			else:
				t := try item.time.shift(nanoseconds(item.delta))

				if kind == "pipeline":
					u := try parse_timestamp(t.to_string())
					if u.seconds != t.seconds or u.nanos != t.nanos:
						return error(Error("round-trip mismatch"))

				checksum += t.seconds + t.nanos

	return ok(checksum)

fn main() Unit!:
	args := arguments()
	kind := args[0]
	count := args[1].parse_int().get()
	rounds := args[2].parse_int().get()
	data := inputs(kind, count)
	warmup := try work(kind, data, 1)
	start := monotonic()
	checksum := try work(kind, data, rounds)
	elapsed := monotonic().difference(start).nanos
	if checksum != warmup * rounds: return error(Error("checksum changed"))

	print("{kind} {count} {rounds} {elapsed} {checksum}")
