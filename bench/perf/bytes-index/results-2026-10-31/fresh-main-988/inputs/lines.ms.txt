import stdout from "@std/io/io.ms"
import clock from "@std/time.ms"

fn sum_lines(text: String) Int:
    lines := text.lines()
    var checksum = lines.length() * 17
    for line in lines:
        checksum += line.byte_length()
    return checksum

fn measure(name: String, text: String, expected: Int, iterations: Int) Int:
    if sum_lines(text) != expected: return 1

    var checksum = 0
    started := clock().now()
    for iteration in 0..iterations:
        checksum += sum_lines(text)

    elapsed := clock().now().difference(started).to_nanoseconds()
    if checksum != expected * iterations:
        return 1

    _written := stdout().write(
        "{name},{text.byte_length()},{iterations},{elapsed},{checksum}\n"
    )
    return 0

fn main() Int:
    _header := stdout().write("case,input_bytes,iterations,elapsed_ns,checksum\n")
    if measure("empty", "", 0, 1000000) != 0: return 1
    if measure("tiny", "a\nb\n", 36, 1000000) != 0: return 1
    if measure("short_unterminated", "hello", 22, 1000000) != 0: return 1
    if measure("empty_lines", "\n\n\n\n", 68, 1000000) != 0: return 1

    for width in [7, 8, 15, 16, 31, 32, 128, 1024, 8192, 65536]:
        count := 1048576 / (width + 1)
        text := ("x".repeat(width).get() + "\n").repeat(count).get()
        if measure("lf_{width}", text, (width + 17) * count, 64) != 0: return 1

    unicode := ("é日😀".repeat(1000).get() + "\r\n").repeat(128).get()
    if measure("unicode_crlf", unicode, (9000 + 17) * 128, 64) != 0: return 1

    no_newline := "x".repeat(1048576).get()
    if measure("no_newline", no_newline, 1048576 + 17, 64) != 0: return 1

    return 0
