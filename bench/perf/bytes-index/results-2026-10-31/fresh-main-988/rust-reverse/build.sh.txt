#!/usr/bin/env bash
set -euo pipefail
cd /workspace/reverse-discovery
compiler=/workspace/perf-toolchain/rustup/toolchains/1.99.1-x86_64-unknown-linux-gnu/bin/rustc

sha256sum reverse-control.rs rust-reverse-control build.sh > SHA256SUMS
