//! Standalone, dependency-free control for reverse-full.ms (83 cases).
//! Reversal means Unicode scalar order, including combining marks and ZWJ.
use std::hint::black_box;
use std::io::{self, Write};
use std::time::Instant;

struct Case {
    name: String,
    text: String,
    expected: String,
}

fn iterations_for(size: usize) -> i64 {
    if size <= 32 {
        return 1_000_000;
    }
    if size >= 262_144 {
        return 16;
    }
    if size >= 4_096 {
        return 1_024;
    }
    100_000
}

// An owned String is constructed and fully compared on EVERY invocation.
// The function boundary and opaque arguments keep the operation observable.
#[inline(never)]
fn reverse_sum(text: &str, expected: &str) -> i64 {
    let output: String = text.chars().rev().collect();
    if black_box(output.as_str()) != expected {
        return -1;
    }
    output.len() as i64 + 17
}

fn cases() -> Vec<Case> {
    let mut cases = Vec::with_capacity(83);
    for (name, forward, backward) in [
        ("ascii_mixed", "abcDE09_", "_90EDcba"),
        ("ascii_controls", "a\u{0}b\tc\nd!", "!d\nc\tb\u{0}a"),
    ] {
        for length in [0, 1, 2, 7, 8, 15, 16, 17, 31, 32, 63, 64, 128, 4096, 1048576] {
            let copies = length / 8;
            let tail = length % 8;
            // The two patterns are ASCII, so these byte and scalar slices agree.
            let text = forward.repeat(copies) + &forward[..tail];
            let expected = backward[8 - tail..8].to_owned() + &backward.repeat(copies);
            cases.push(Case { name: format!("{name}_n{length}"), text, expected });
        }
    }
    for (name, forward, backward) in [
        ("two_byte", "éΩüß", "ßüΩé"),
        ("three_byte", "日漢€☃", "☃€漢日"),
        ("four_byte", "😀🙂🚀🌍", "🌍🚀🙂😀"),
        ("mixed", "aé日😀\u{0}e\u{301}\u{fffd}", "\u{fffd}\u{301}e\u{0}😀日éa"),
        ("combining", "e\u{301}a\u{308}", "\u{308}a\u{301}e"),
        ("zwj", "👩\u{200d}🚀", "🚀\u{200d}👩"),
        ("replacement", "\u{fffd}", "\u{fffd}"),
    ] {
        for copies in [1, 2, 8, 32, 256, 4096, 65536] {
            cases.push(Case {
                name: format!("{name}_copies{copies}"),
                text: forward.repeat(copies),
                expected: backward.repeat(copies),
            });
        }
    }
    for length in [4096, 1048576] {
        let ascii = "x".repeat(length);
        cases.push(Case {
            name: format!("late_unicode_{length}"),
            text: ascii.clone() + "é日😀",
            expected: "😀日é".to_owned() + &ascii,
        });
        cases.push(Case {
            name: format!("early_unicode_{length}"),
            text: "é日😀".to_owned() + &ascii,
            expected: ascii + "😀日é",
        });
    }
    assert_eq!(cases.len(), 83, "incomplete corpus");
    cases
}

fn measure(case: &Case, output: &mut impl Write) -> io::Result<()> {
    let expected_sum = case.expected.len() as i64 + 17;
    assert_eq!(reverse_sum(black_box(&case.text), black_box(&case.expected)), expected_sum,
        "initial output mismatch for {}", case.name);
    let iterations = iterations_for(case.text.len());
    let mut checksum = 0_i64;
    let started = Instant::now();
    for _ in 0..iterations {
        checksum += black_box(reverse_sum(black_box(&case.text), black_box(&case.expected)));
    }
    let elapsed = started.elapsed().as_nanos();
    assert_eq!(checksum, expected_sum * iterations, "checksum mismatch for {}", case.name);
    writeln!(output, "{},{},{},{},{}", case.name, case.text.len(), iterations, elapsed, checksum)
}

fn main() -> io::Result<()> {
    let mode = std::env::args().nth(1);
    assert!(std::env::args().len() <= 2, "expected zero or one argument");
    let corpus = cases(); // All input/expected construction precedes ALL clocks.
    let mut output = io::BufWriter::new(io::stdout().lock());
    match mode.as_deref() {
        Some("--verify") => {
            for case in &corpus {
                assert_eq!(reverse_sum(black_box(&case.text), black_box(&case.expected)),
                    case.expected.len() as i64 + 17, "output mismatch for {}", case.name);
            }
            writeln!(output, "verified {} complete scalar-reversal outputs", corpus.len())?;
        }
        Some("--inventory") => {
            writeln!(output, "case,input_bytes,iterations,checksum")?;
            for case in &corpus {
                let iterations = iterations_for(case.text.len());
                writeln!(output, "{},{},{},{}", case.name, case.text.len(), iterations,
                    (case.expected.len() as i64 + 17) * iterations)?;
            }
        }
        None => {
            writeln!(output, "case,input_bytes,iterations,elapsed_ns,checksum")?;
            for case in &corpus {
                measure(case, &mut output)?;
            }
        }
        Some(other) => panic!("unknown mode: {other}"),
    }
    output.flush()
}
