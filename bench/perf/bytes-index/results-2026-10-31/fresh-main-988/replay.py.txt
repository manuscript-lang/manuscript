import hashlib
import json
from pathlib import Path
import subprocess

root = Path('/workspace/manuscript')
out = Path(__file__).resolve().parent
digest = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
record = {'complete': False, 'commands': []}
record['inputs'] = {}
for p in [out/'build-record.json', out/'production.patch', out/'replay.py',
          root/'bench/perf/bytes-index/compare.py', root/'bench/perf/bytes-index/identities.json',
          out/'runner/string-normalize-ascii/compare_matrix.py', out/'runner/string-lines/compare.py']:
    record['inputs'][str(p)] = digest(p)
for suite in ['lines', 'normalize', 'reverse']:
    for suffix in ['programs', 'identities']:
        p = out/(suite+'-'+suffix+'.json')
        record['inputs'][str(p)] = digest(p)
record['programs'] = json.loads((out/'build-record.json').read_text())['programs']
for label, path in {'rust-byte':'/workspace/perf-toolchain/bytes-index-rust-target/release/bytes-index-rust',
                    'rust-reverse':'/workspace/reverse-discovery/rust-reverse-control'}.items():
    record['programs'][label] = {'path':path,'sha256':digest(path)}
def save():
    (out/'replay-record.json').write_text(json.dumps(record,indent=2)+'\n')
save()
def run(args, name):
    print(name+' starting',flush=True)
    with (out/(name+'.log')).open('wb') as log:
        result=subprocess.run(args,cwd=root,stdout=log,stderr=subprocess.STDOUT)
    record['commands'].append({'argv':args,'exit':result.returncode,'log':name+'.log'})
    save()
    result.check_returncode()
    print(name+' complete',flush=True)
run(['python3',str(root/'bench/perf/bytes-index/compare.py'),str(out/'baseline-byte'),str(out/'word-byte'),
     record['programs']['rust-byte']['path'],str(out/'byte220'),'--cpu','0'],'byte220')
for suite in ['lines','normalize','reverse']:
    args=['python3',str(out/'runner/string-normalize-ascii/compare_matrix.py'),str(out/(suite+'-programs.json')),
          str(out/(suite+'-identities.json')),str(out/(suite+'-matrix')),'--cpu','0']
    if suite!='reverse':args += ['--cycles','2']
    run(args,suite)
record['integrity_after']={p:digest(p)==h for p,h in record['inputs'].items()}
record['programs_integrity_after']={k:digest(v['path'])==v['sha256'] for k,v in record['programs'].items()}
assert all(record['integrity_after'].values()) and all(record['programs_integrity_after'].values())
record['complete']=True
save()
