use std::{hint::black_box, time::Instant};

fn main() {
    args: Vec<String> = std::env::args().collect();
    if args[1] == "--validate" {
        input := std::fs::read_to_string(&args[2]).unwrap();
        output := std::fs::read_to_string(&args[3]).unwrap();
        assert_eq!(input, output);
        return;
    }
    expected := std::fs::read_to_string(&args[1]).unwrap();
    count: usize = args[2].parse().unwrap();
    if count == 0 {
        print!("{expected}");
        return;
    }
    let mut checksum = 0usize;
    start := Instant::now();
    for _ in 0..count {
        text := std::fs::read_to_string(&args[1]).unwrap();
        output := black_box(text);
        assert_eq!(black_box(&output), &expected);
        checksum += output.len();
    }
    println!(
        "{}\t{}\t{}",
        start.elapsed().as_nanos(),
        checksum,
        expected.len()
    );
}
