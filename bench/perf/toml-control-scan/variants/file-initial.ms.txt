import parse, stringify from "@std/data/toml.ms"
import arguments from "@std/os/process.ms"
import filesystem from "@std/os/fs.ms"
import stdout from "@std/io/io.ms"
import clock from "@std/time.ms"

fn main() Int:
    args := arguments()
    path := args.get(0)
    text := filesystem().read_file(path).get().decode_utf8().get()
    count := args.get(1).parse_int().get()
    expected := stringify(parse(text).get()).get()

    if count == 0:
        stdout().write(expected).get()
        return 0

    var checksum = 0
    start := clock().now()

    for unused in 0..count:
        current := filesystem().read_file(path).get().decode_utf8().get()
        output := stringify(parse(current).get()).get()
        if output != expected: return 1

        checksum += output.byte_length()

    elapsed := clock().now().difference(start).to_nanoseconds()
    stdout().write("{elapsed}\t{checksum}\t{expected.byte_length()}\n").get()
    return 0
