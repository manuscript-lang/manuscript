import parse from "@std/data/toml.ms"
import arguments from "@std/os/process.ms"
import filesystem from "@std/os/fs.ms"
import stdout from "@std/io/io.ms"
import clock from "@std/os/time.ms"

fn diagnostic(text: String) String:
    match parse(text):
        ok(_): return "accepted"
        error(cause): return cause.message()

export fn main() Int:
    let args = arguments()
    let path = args.get(0)
    let text = filesystem().read_file(path).get().decode_utf8().get()
    let count = args.get(1).parse_int().get()
    let expected = diagnostic(text)

    if count == 0:
        stdout().write(expected).get()
        return 0

    var checksum = 0
    let start = clock().now()

    for unused in 0..count:
        let current = filesystem().read_file(path).get().decode_utf8().get()
        let output = diagnostic(current)
        if output != expected: return 1

        checksum += output.byte_length()

    let elapsed = clock().now().difference(start).to_nanoseconds()
    stdout().write("{elapsed}\t{checksum}\t{expected.byte_length()}\n").get()
    return 0
