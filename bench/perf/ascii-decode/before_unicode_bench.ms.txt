import ascii_decode, EncodingError from "./before.ms"
import keep, measure from "@std/test/bench.ms"
import expect_equal from "@std/test/test.ms"

bench "ascii.unicode.accent.1":
    data := (try "é".repeat(1)).to_bytes()
    expected: String!EncodingError = error(EncodingError.InvalidData)
    try expect_equal(ascii_decode(data), expected)

    fn work() Unit!:
        try expect_equal(ascii_decode(data), expected)
        return keep(data.length())

    try measure(1, work)
