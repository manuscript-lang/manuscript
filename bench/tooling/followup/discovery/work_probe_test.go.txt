var physicalWalks, physicalVisits, physicalReads int
func discoveryProbeWalkDir(root string,visit fs.WalkDirFunc) error {
 physicalWalks++
 return filepath.WalkDir(root,func(path string,d fs.DirEntry,err error) error {physicalVisits++;return visit(path,d,err)})
}
func discoveryProbeReadFile(path string)([]byte,error){physicalReads++;return os.ReadFile(path)}

 for _,aliases:=range []int{1,2,4}{
