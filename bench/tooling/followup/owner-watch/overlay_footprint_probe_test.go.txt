package lsp

import (
 "bytes"
 "os"
 "path/filepath"
 "strings"
 "testing"
)

func TestWatchCrossOwnerOverlayFootprints(t *testing.T){
 for _,directory:=range []bool{false,true}{for _,retarget:=range []bool{false,true}{
  label:="file";if directory{label="directory"};if retarget{label+="-retarget"}else{label+="-delete"}
  t.Run(label,func(t *testing.T){
   a,_:=snapshotFiles(t,map[string]string{"ms.toml":"[project]\nname = \"a\"\nsource = \".\"\nentry = \"main.ms\"\n","main.ms":"fn a() Int: 0\n"})
   b,files:=snapshotFiles(t,map[string]string{"ms.toml":"[project]\nname = \"b\"\nsource = \".\"\nentry = \"main.ms\"\n","main.ms":"import value from \"./subdir/dep.ms\"\nfn b() Int: value()\n","subdir/dep.ms":"export fn value() Int: 1\n"})
   route:=filepath.Join(a,"link.ms");target:=filepath.Join(b,"subdir/dep.ms")
   if directory{route=filepath.Join(a,"alias.dir");target=filepath.Join(b,"subdir")}
   if err:=os.Symlink(target,route);err!=nil{t.Skipf("symlinks unavailable: %v",err)}
   sourcePath:=route;if directory{sourcePath=filepath.Join(route,"dep.ms")}
   open:=document{URI:fileURI(sourcePath),Text:"export fn value() String: \"unsaved\"\n",Version:1}
   c:=performanceProject(t,1,1);c.Text=strings.Replace(c.Text,"    let result","    let unused = 1\n    let result",1)
   main:=files["main.ms"];s:=snapshotServer(open,main,c)
   before,retained:=s.checkedDocument(main),s.checkedDocument(c)
   if before.Success(){t.Fatal("cross-owner physical overlay was not admitted",before.Diagnostics)}
   index,lint:=s.documentIndex(retained),s.programLint(retained)
   if len(index)==0||len(lint)==0{t.Fatal("missing C derived-state fixture")}
   if _,cached:=s.cache[open.URI];cached{t.Fatal("A overlay should be unrequested before mutation")}
   if err:=os.Remove(route);err!=nil{t.Fatal(err)}
   if retarget{other:=t.TempDir();watchWrite(t,filepath.Join(other,"dep.ms"),"export fn value() Int: 7\n");target=filepath.Join(other,"dep.ms");if directory{target=other};if err:=os.Symlink(target,route);err!=nil{t.Fatal(err)}}
   var output bytes.Buffer;s.output=&output
   watchSend(t,s,fileURI(route))
   got:=snapshotEquivalent(t,s,main)
   if !got.Success()||got.Program==before.Program{t.Fatal("old cross-owner overlay footprint retained stale B failure",got.Diagnostics)}
   if s.checkedDocument(c).Program!=retained.Program||&s.documentIndex(retained)[0]!=&index[0]||&s.programLint(retained)[0]!=&lint[0]{t.Fatal("cross-owner A/B event evicted independent C derived identities")}
   for _,message:=range readMessages(t,&output){if message["params"].(map[string]any)["uri"]==c.URI{t.Fatal("cross-owner event republished C")}}
  })
 }}
}
