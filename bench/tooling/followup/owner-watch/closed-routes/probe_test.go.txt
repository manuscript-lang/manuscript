package lsp

import (
 "os"
 "path/filepath"
 "strings"
 "testing"
)

// External observation control. Pass requires the exact stale/fresh disagreement.
func TestReviewClosedLexicalAliasOutsideOwner(t *testing.T) {
 for _, mode := range []string{"relative_import", "manifest_alias", "manifest_entry"} {
  for _, independentOpen := range []bool{false,true} {
   name := mode
   if independentOpen { name += "_independent_A_open" }
   t.Run(name, func(t *testing.T) {
    parent := t.TempDir()
    a, b := filepath.Join(parent,"A"),filepath.Join(parent,"B")
    alias := filepath.Join(a,"link.ms")
    valueSource := "export fn value() Int: 1\n"
    mainSource := "import value from \"../A/link.ms\"\nfn main() Int: value()\n"
    manifest := "[project]\nname = \"B\"\nsource = \".\"\nentry = \"main.ms\"\n"
    requested := "main.ms"
    if mode == "manifest_alias" {
     mainSource = "import value from \"@route\"\nfn main() Int: value()\n"
     manifest += "[aliases]\n\"@route\" = \"../A/link.ms\"\n"
    }
    if mode == "manifest_entry" {
     valueSource = "fn main() Int: 1\n"
     manifest = "[project]\nname = \"B\"\nsource = \".\"\nentry = \"../A/link.ms\"\n"
     requested = "dep.ms"
    }
    watchWrite(t,filepath.Join(b,"ms.toml"),manifest)
    watchWrite(t,filepath.Join(b,"main.ms"),mainSource)
    watchWrite(t,filepath.Join(b,"dep.ms"),valueSource)
    watchWrite(t,filepath.Join(a,"main.ms"),"fn independent() Int: 2\n")
    if err:=os.Symlink(filepath.Join(b,"dep.ms"),alias);err!=nil {t.Fatal(err)}
    text,err:=os.ReadFile(filepath.Join(b,requested));if err!=nil {t.Fatal(err)}
    d:=document{URI:fileURI(filepath.Join(b,requested)),Text:string(text),Version:1}
    s:=snapshotServer(d)
    if independentOpen {
     ad:=document{URI:fileURI(filepath.Join(a,"main.ms")),Text:"fn independent() Int: 2\n",Version:1}
     s.docs[ad.URI]=ad
     if got:=s.checkedDocument(ad);!got.Success(){t.Fatal("invalid A control",got.Diagnostics)}
    }
    before:=s.checkedDocument(d)
    if !before.Success()||before.Program==nil {t.Fatal("closed lexical return-inside alias was not valid",before.Diagnostics)}
    modules:=[]string{}
    for _,m:=range before.Program.Modules {modules=append(modules,m.File)}
    if err:=os.Remove(alias);err!=nil {t.Fatal(err)}
    watchWrite(t,alias,"export fn value() Int: 9\n")
    fresh:=analyze(d,s.docs)
    if fresh.Success(){t.Fatal("fresh analysis did not reject lexical escape after alias replacement")}
    hasEscape:=false
    for _,e:=range fresh.Diagnostics {hasEscape=hasEscape||strings.Contains(e.Message,"escapes project root")}
    if !hasEscape {t.Fatal("fresh error was not intended confinement diagnostic",fresh.Diagnostics)}
    watchSend(t,s,fileURI(alias))
    cached:=s.checkedDocument(d)
    if !cached.Success()||cached.Program!=before.Program {t.Fatal("expected stale success observation changed; reassess control",cached.Diagnostics)}
    t.Logf("CONFIRMED mode=%s independent_A_open=%v initial_success=true canonical_modules=%v cached_same_program=true cached_success=true fresh_success=false fresh_diagnostics=%v",mode,independentOpen,modules,fresh.Diagnostics)
   })
  }
 }
}
