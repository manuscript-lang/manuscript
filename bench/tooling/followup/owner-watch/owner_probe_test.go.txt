package lsp

import (
 "bytes"
 "encoding/json"
 "fmt"
 "os"
 "path/filepath"
 "testing"

 "github.com/manuscript-lang/manuscript/pkg/compiler"
 "github.com/manuscript-lang/manuscript/pkg/project"
)
var probeCompiles,probeManifests,probeCanonical int
func ownerProbeCompile(path string,options compiler.Options) compiler.Result {probeCompiles++;return compiler.CompileFile(path,options)}
func ownerProbeManifest(path string)(string,bool){probeManifests++;return project.FindManifest(path)}
func ownerProbeCanonical(path string)(string,error){probeCanonical++;return filepath.EvalSymlinks(path)}
type ownerProbeOutput struct{publications int}
func(w *ownerProbeOutput)Write(p []byte)(int,error){if bytes.Contains(p,[]byte(`"method":"textDocument/publishDiagnostics"`)){w.publications++};return len(p),nil}
func BenchmarkOwnerEvent(b *testing.B){for _,roots:=range []int{2,4,8}{b.Run(fmt.Sprintf("roots=%d",roots),func(b *testing.B){ownerProbeBenchmark(b,roots,false)})}}
func BenchmarkOrdinaryChange(b *testing.B){for _,roots:=range []int{2,4,8}{b.Run(fmt.Sprintf("roots=%d",roots),func(b *testing.B){ownerProbeBenchmark(b,roots,true)})}}
func ownerProbeBenchmark(b *testing.B,roots int,ordinary bool){
 b.StopTimer()
 docs:=make([]document,roots)
 for n:=range docs {docs[n]=performanceProject(b,4,8)}
 s:=snapshotServer(docs...);out:=&ownerProbeOutput{};s.output=out
 for _,d:=range docs {if checked:=s.checkedDocument(d);!checked.Success(){b.Fatal(checked.Diagnostics)}}
 dep:=filepath.Join(filepath.Dir(filename(docs[0].URI)),"part_0.ms")
 original,err:=os.ReadFile(dep);if err!=nil {b.Fatal(err)}
 watchParams,_:=json.Marshal(map[string]any{"changes":[]any{map[string]any{"uri":fileURI(dep),"type":2}}})
 var compiles,publications,manifests,canonical,retained int
 b.ReportAllocs();b.ResetTimer()
 for n:=0;n<b.N;n++{
  d:=s.docs[docs[0].URI]
  r:=&request{Method:"workspace/didChangeWatchedFiles",Params:watchParams}
  if ordinary{
   d.Version++
   params,_:=json.Marshal(map[string]any{"textDocument":d,"contentChanges":[]any{map[string]any{"text":docs[0].Text+fmt.Sprintf("# edit%d\n",n)}}})
   r=&request{Method:"textDocument/didChange",Params:params};if err:=json.Unmarshal(params,&r.params);err!=nil{b.Fatal(err)}
  }else{
   source:=original;if n%2==0 {source=[]byte("export fn f_0_0(value: Int) String: \"changed\"\n")}
   if err:=os.WriteFile(dep,source,0600);err!=nil{b.Fatal(err)}
  }
  before:=s.checkedDocument(docs[1]).Program
  probeCompiles,probeManifests,probeCanonical=0,0,0;out.publications=0
  b.StartTimer()
  _,rpcErr,err:=s.handle(r)
  b.StopTimer()
  if err!=nil||rpcErr!=nil{b.Fatal(err,rpcErr)}
  // Save counts before uncached comparison; validation never contributes metrics.
  compiles+=probeCompiles;publications+=out.publications;manifests+=probeManifests;canonical+=probeCanonical
  if s.checkedDocument(docs[1]).Program==before{retained++}
  checked:=s.checkedDocument(s.docs[docs[0].URI])
  if ordinary && !checked.Success(){b.Fatal("ordinary edit lost valid source",checked.Diagnostics)}
  if !ordinary && checked.Success()!=(n%2!=0){b.Fatal("watch failed to refresh actual changed dependency",n,checked.Diagnostics)}
  // Fresh compiler result and every ordinary document remain complete.
  fresh:=analyze(s.docs[docs[0].URI],s.docs)
  if checked.Success()!=fresh.Success()||len(checked.Diagnostics)!=len(fresh.Diagnostics){b.Fatal("cached watch/edit differs from fresh check",checked.Diagnostics,fresh.Diagnostics)}
 }
 b.ReportMetric(float64(compiles)/float64(b.N),"compiles/op");b.ReportMetric(float64(publications)/float64(b.N),"pubs/op")
 b.ReportMetric(float64(manifests)/float64(b.N),"manifest/op");b.ReportMetric(float64(canonical)/float64(b.N),"canonical/op");b.ReportMetric(float64(retained)/float64(b.N),"B-retained/op")
}
