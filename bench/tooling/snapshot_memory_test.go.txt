package lsp

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"runtime"
	"sort"
	"strconv"
	"strings"
	"testing"

	"github.com/manuscript-lang/manuscript/pkg/compiler"
)

// External overlay probe: this uses only pre-existing server APIs, so the exact
// same workload runs against stage 2 and the shared-snapshot implementation.
func TestSnapshotMemoryEvidence(t *testing.T) {
	root := t.TempDir()
	if err := os.WriteFile(filepath.Join(root, "ms.toml"), []byte("[project]\nname = \"snapshot_memory\"\nsource = \".\"\n"), 0600); err != nil {
		t.Fatal(err)
	}
	all := make([]document, 10)
	for module := range all {
		var source strings.Builder
		for fn := 0; fn < 100; fn++ {
			fmt.Fprintf(&source, "fn f%03d(value: Int) Int: value + %d\n", fn, fn)
		}
		path := filepath.Join(root, fmt.Sprintf("module%02d.ms", module))
		all[module] = document{URI: fileURI(path), Text: source.String(), Version: 1}
		if err := os.WriteFile(path, []byte(all[module].Text), 0600); err != nil {
			t.Fatal(err)
		}
	}
	func() {
		warm := analyze(all[0], nil)
		if !warm.Success() || len(warm.Program.Modules) != 10 {
			t.Fatal("invalid memory fixture", warm.Diagnostics)
		}
	}()
	s := &server{docs: map[string]document{}, cache: map[string]compiler.Result{}}
	type sample struct {
		HeapBytes  uint64 `json:"heap_bytes"`
		Objects    uint64 `json:"objects"`
		CachedURIs int    `json:"cached_uris"`
		Programs   int    `json:"programs"`
	}
	measure := func() sample {
		runtime.GC()
		var stats runtime.MemStats
		runtime.ReadMemStats(&stats)
		programs := map[any]bool{}
		for _, result := range s.cache {
			programs[result.Program] = true
		}
		runtime.KeepAlive(s)
		return sample{stats.HeapAlloc, stats.HeapObjects, len(s.cache), len(programs)}
	}
	open := func(d document) {
		s.docs[d.URI] = d
		s.invalidate()
		uris := make([]string, 0, len(s.docs))
		for uri := range s.docs {
			uris = append(uris, uri)
		}
		sort.Strings(uris)
		for _, uri := range uris {
			result := s.checkedDocument(s.docs[uri])
			if !result.Success() || len(result.Program.Modules) != 10 {
				t.Fatal("lost checked library modules", result.Diagnostics)
			}
		}
	}
	baseline := measure()
	open(all[0])
	one := measure()
	for _, d := range all[1:] {
		open(d)
	}
	ten := measure()
	if ten.CachedURIs != 10 {
		t.Fatalf("memory fixture cached only %d open documents", ten.CachedURIs)
	}
	if expected := os.Getenv("MANUSCRIPT_SNAPSHOT_EXPECT_PROGRAMS"); expected != "" {
		want, err := strconv.Atoi(expected)
		if err != nil || ten.Programs != want {
			t.Fatalf("retained %d programs, expected %q: %v", ten.Programs, expected, err)
		}
	}
	s.docs = map[string]document{}
	s.invalidate()
	closed := measure()
	if closed.CachedURIs != 0 || closed.Programs != 0 {
		t.Fatal("close/invalidation retained URI result aliases")
	}
	encoded, err := json.Marshal(map[string]any{
		"baseline": baseline, "one_open": one, "ten_open": ten, "all_closed": closed,
		"one_minus_base":    int64(one.HeapBytes) - int64(baseline.HeapBytes),
		"ten_minus_base":    int64(ten.HeapBytes) - int64(baseline.HeapBytes),
		"ten_minus_one":     int64(ten.HeapBytes) - int64(one.HeapBytes),
		"closed_minus_base": int64(closed.HeapBytes) - int64(baseline.HeapBytes),
	})
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("snapshot_retention=%s", encoded)
	runtime.KeepAlive(all)
	runtime.KeepAlive(s)
}
